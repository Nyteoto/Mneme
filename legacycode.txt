void playTone(int freq, int durationMs, float volume) {
if (!initAudio()) return;

volume = constrain(volume, 0.0, 0.2) * globalVolume;

int samples = (durationMs * SAMPLE_RATE) / 1000;
float phaseIncrement = (2.0 * PI * freq) / SAMPLE_RATE;
float currentPhase = 0.0;

for (int i = 0; i < samples; i++) {
float sample = sin(currentPhase) * volume;
int16_t audioSample = sample * 32767;

```
audio.write(audioSample);
audio.write(audioSample);

currentPhase += phaseIncrement;
if (currentPhase >= 2.0 * PI) {
  currentPhase -= 2.0 * PI;
}

```

}

void saveGameData() {
EEPROM.put(MAGIC_NUMBER_ADDR, MAGIC_NUMBER);
EEPROM.put(DATA_VERSION_ADDR, CURRENT_DATA_VERSION);
EEPROM.put(LEVEL_ADDR, levelCounter);
EEPROM.put(EXP_ADDR, currentExp);
EEPROM.put(EXP_REQ_ADDR, expForNextLevel);
EEPROM.put(BUTTON_COUNT_ADDR, buttonPressCount);
EEPROM.put(AUDIO_VOLUME_ADDR, globalVolume);

// Save all log slots (including zeros for empty slots)
for (int i = 0; i < 3; i++) {
EEPROM.put(TASK_TIMER_LOGS_ADDR + (i * 4), taskPriorityLogs[i]);
}
EEPROM.put(TASK_TIMER_LOGS_ADDR + 12, taskPriorityLogCount);

DateTime now = rtc.now();
int currentDay = now.day();
EEPROM.put(LAST_DAY_ADDR, currentDay);

EEPROM.put(HEATMAP_LAST_DAY_ADDR, heatmapLastDay);
EEPROM.put(HEATMAP_STREAK_COUNT_ADDR, heatmapStreakCount);

uint32_t gridBitfield = 0;
for (int i = 0; i < 21; i++) {
if (heatmapGrid[i]) {
gridBitfield |= (1UL << i);
}
}
EEPROM.put(HEATMAP_GRID_DATA_ADDR, gridBitfield);

// NEW: Save if heatmap was logged today
EEPROM.put(HEATMAP_LOGGED_TODAY_ADDR, heatmapAlreadyLoggedToday);

EEPROM.commit();
lastSaveTime = millis();
}

void loadGameData() {
uint16_t magic;
uint8_t version;
EEPROM.get(MAGIC_NUMBER_ADDR, magic);

if (magic == MAGIC_NUMBER) {
EEPROM.get(DATA_VERSION_ADDR, version);

```
EEPROM.get(LEVEL_ADDR, levelCounter);
EEPROM.get(EXP_ADDR, currentExp);
EEPROM.get(EXP_REQ_ADDR, expForNextLevel);
EEPROM.get(BUTTON_COUNT_ADDR, buttonPressCount);
EEPROM.get(LAST_DAY_ADDR, lastSavedDay);

for (int i = 0; i < 3; i++) {
  EEPROM.get(TASK_TIMER_LOGS_ADDR + (i * 4), taskPriorityLogs[i]);
}
EEPROM.get(TASK_TIMER_LOGS_ADDR + 12, taskPriorityLogCount);

// Load audio volume
if (version >= 4) {
  EEPROM.get(AUDIO_VOLUME_ADDR, globalVolume);
} else {
  globalVolume = 0.2f; // Default 20%
}

// FIX: Filter out zero-value logs and adjust log count accordingly
int validLogCount = 0;
for (int i = 0; i < 3; i++) {
  if (taskPriorityLogs[i] > 0) {
    // This is a valid log with actual time data
    if (i != validLogCount) {
      // Move valid log to the front
      taskPriorityLogs[validLogCount] = taskPriorityLogs[i];
      taskPriorityLogs[i] = 0;
    }
    validLogCount++;
  }
}
taskPriorityLogCount = validLogCount;

if (version >= 3) {
  EEPROM.get(HEATMAP_LAST_DAY_ADDR, heatmapLastDay);
  EEPROM.get(HEATMAP_STREAK_COUNT_ADDR, heatmapStreakCount);

  uint32_t gridBitfield;
  EEPROM.get(HEATMAP_GRID_DATA_ADDR, gridBitfield);
  for (int i = 0; i < 21; i++) {
    heatmapGrid[i] = (gridBitfield & (1UL << i)) != 0;
  }

  // NEW: Load if heatmap was logged today (only for version 5+)
  if (version >= 5) {
    EEPROM.get(HEATMAP_LOGGED_TODAY_ADDR, heatmapAlreadyLoggedToday);
  } else {
    // For older versions, initialize as false
    heatmapAlreadyLoggedToday = false;
    // Update data version
    EEPROM.put(DATA_VERSION_ADDR, CURRENT_DATA_VERSION);
  }
} else {
  heatmapLastDay = -1;
  heatmapStreakCount = 0;
  for (int i = 0; i < 21; i++) {
    heatmapGrid[i] = false;
  }
  heatmapAlreadyLoggedToday = false;
}

if (version != CURRENT_DATA_VERSION) {
  EEPROM.put(DATA_VERSION_ADDR, CURRENT_DATA_VERSION);
  EEPROM.commit();
}

```

} else {
// No saved data found, initializing defaults
levelCounter = 1;
currentExp = 0.0f;
expForNextLevel = 10.0f;
buttonPressCount = 0;
globalVolume = 0.2f;
for (int i = 0; i < 3; i++) {
taskPriorityLogs[i] = 0;
}
taskPriorityLogCount = 0;
heatmapLastDay = -1;
heatmapStreakCount = 0;
for (int i = 0; i < 21; i++) {
heatmapGrid[i] = false;
}
heatmapAlreadyLoggedToday = false;

void checkHeatmapStreak(int currentDay) {
if (heatmapLastDay != -1) {
int dayDifference = (currentDay - heatmapLastDay + 31) % 31;
if (dayDifference > 1) {
resetHeatmapGrid();
}
}

// NEW: Reset the logged today flag on day change
heatmapAlreadyLoggedToday = false;
}
void resetHeatmapGrid() {
heatmapStreakCount = 0;
for (int i = 0; i < 21; i++) {
heatmapGrid[i] = false;
}
heatmapAlreadyLoggedToday = false;  // Also reset the daily flag
saveGameData();
}

void updateLevelUpAnimation() {
if (!levelUpActive) return;

unsigned long elapsed = millis() - levelUpStartTime;
if (elapsed >= LEVEL_UP_DURATION) {
levelUpActive = false;
return;
}

float intensity = 1.0f;
if (elapsed < 500) {
intensity = (elapsed / 100) % 2 == 0 ? 1.0f : 0.0f;
} else if (elapsed < 1500) {
intensity = (elapsed / 200) % 2 == 0 ? 1.0f : 0.0f;
} else if (elapsed < 4500) {
intensity = 1.0f;
} else {
float fadeProgress = (elapsed - 4500.0f) / 4000.0f;
intensity = 1.0f - fadeProgress;
}

if (intensity > 0.1f) {
drawLevelUpBar(intensity);
}
}

// OPTIMIZED: Read battery voltage with Arduino-compatible ADC control
float readBatteryVoltageOptimized() {
const int NUM_SAMPLES = 8;  // Reduced from 16 for speed
long totalAdcValue = 0;

// Arduino analogRead already initializes ADC
for(int i = 0; i < NUM_SAMPLES; i++) {
delayMicroseconds(100);  // Shorter delay
totalAdcValue += analogRead(BATT_ADC_PIN);
}

int adcValue = totalAdcValue / NUM_SAMPLES;
float adcVoltage = (adcValue / ADC_MAX_VALUE) * 3.3f;

// Arduino doesn't have adc_deinit, but we can set pin to INPUT
// to reduce leakage through ADC pin
pinMode(BATT_ADC_PIN, INPUT);

return adcVoltage * VOLTAGE_DIVIDER_RATIO;
}

void addExp(float expAmount) {
currentExp += expAmount;

bool leveledUp = false;
while (currentExp >= expForNextLevel) {
currentExp -= expForNextLevel;
levelCounter++;
leveledUp = true;
expForNextLevel = calculateExpForLevel(levelCounter);
startLevelUpAnimation();
}

if (leveledUp) saveGameData();
}

float calculateExpForLevel(int level) {
float base = 10.0f;
float growth = log(level + 5.0f);
float expRequired = base * growth;
return round(expRequired * 100.0f) / 100.0f;
}
